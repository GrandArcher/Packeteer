-- Packeteer MIB. Read-only status of a Packeteer controller and the SNMP
-- notifications it sends for its events.
--
-- The agent is the snmpagent exporter and the notifications come from the
-- snmptrap notifier (docs/monitoring.md). Both place this module under
-- the enterprise_oid they are configured with. The default is the
-- NET-SNMP experimental arc below; Packeteer has no enterprise number of its
-- own, so for production change the one assignment of "packeteer" to an arc
-- you own and set enterprise_oid to match.

PACKETEER-MIB DEFINITIONS ::= BEGIN

IMPORTS
    MODULE-IDENTITY, OBJECT-TYPE, NOTIFICATION-TYPE, Gauge32, Integer32,
    enterprises
        FROM SNMPv2-SMI
    DisplayString, TruthValue
        FROM SNMPv2-TC
    MODULE-COMPLIANCE, OBJECT-GROUP, NOTIFICATION-GROUP
        FROM SNMPv2-CONF;

packeteer MODULE-IDENTITY
    LAST-UPDATED "202610290000Z"
    ORGANIZATION "Packeteer project"
    CONTACT-INFO
        "https://github.com/GrandArcher/Packeteer"
    DESCRIPTION
        "Status and notifications of a Packeteer BGP performance
         controller. Nothing in this module is writable."
    REVISION "202610290000Z"
    DESCRIPTION
        "First version."
    ::= { enterprises 8072 9999 9999 7717 }

pktrNotifications OBJECT IDENTIFIER ::= { packeteer 0 }
pktrEvent         OBJECT IDENTIFIER ::= { packeteer 1 }
pktrStatus        OBJECT IDENTIFIER ::= { packeteer 2 }
pktrConformance   OBJECT IDENTIFIER ::= { packeteer 4 }

-- removed. Read pktrProviderName to find a provider.
